function getDashboardStats() {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Results");
    if (!sheet) return { status: "success", totalUsers: 0, dominantCounts: {}, recentSubmissions: [] };
    var data = sheet.getDataRange().getDisplayValues();
    if (data.length <= 1) return { status: "success", totalUsers: 0, dominantCounts: {}, recentSubmissions: [] };

    var totalUsers = 0;
    var dominantCounts = {};
    var recentSubmissions = [];

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var domValue = row[11] ? row[11].toString().trim() : "";
      if (!domValue || domValue.startsWith("{")) continue;

      totalUsers++;
      dominantCounts[domValue] = (dominantCounts[domValue] || 0) + 1;
      if (i >= data.length - 10) recentSubmissions.push({ date: row[0], name: row[1], email: row[2], dominant: domValue });
    }
    return { status: "success", totalUsers: totalUsers, dominantCounts: dominantCounts, recentSubmissions: recentSubmissions.reverse() };
  } catch(e) { return { status: "error", message: e.message }; }
}

function getAdministratorDashboardData() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = { status: "success", totalUsers: 0, totalQuestions: 0, usersList: [], questionsList: [] };

    var sheetResults = ss.getSheetByName("Results");
    if (sheetResults) {
      var rValues = sheetResults.getDataRange().getDisplayValues();
      if (rValues.length > 1) { data.usersList = rValues.slice(1).reverse(); data.totalUsers = data.usersList.length; }
    }

    var sheetQuestions = ss.getSheetByName("Questions");
    if (sheetQuestions) {
      var qValues = sheetQuestions.getDataRange().getDisplayValues();
      if (qValues.length > 1) { data.questionsList = qValues.slice(1); data.totalQuestions = data.questionsList.length; }
    }
    return data;
  } catch (e) { return { status: "error", message: e.message }; }
}

function getQuestionsFromDatabase() {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Questions");

    var data = sheet.getDataRange().getDisplayValues();

      if (data[i][0] && data[i][1]) { questionsArray.push({ id: data[i][0].toString().trim(), text: data[i][1].toString().trim(), category: data[i][2] ? data[i][2].toString().trim() : "" }); }
    }
    return questionsArray;

// ✅ อัปเดต: กรองเฉพาะค่า 4 และ 5 ตามคำแนะนำของอาจารย์
// หากผู้ใช้ตอบ 1, 2, หรือ 3 จะส่งค่ากลับเป็น 0 เพื่อไม่ให้นำไปคิดคะแนนรวม
function getQValue(data, index) {
  var val = 0;
  if (data.answersArray && data.answersArray.length >= index) {
    val = Number(data.answersArray[index - 1]) || 0;
  } else {
    val = Number(data['q' + index]) || Number(data['Q' + index]) || Number(data[index]) || 0;
  }
  // นำค่า 1-3 ออก คิดเฉพาะค่า 4 และ 5
  return (val === 4 || val === 5) ? val : 0;
}

function callGeminiAI(prompt, apiKey) {
  try {
    var url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=" + apiKey;
    var payload = { 
      "contents": [{ "parts": [{ "text": prompt }] }], 
      "generationConfig": { "temperature": 0.2, "response_mime_type": "application/json" } 
    };
    var options = { "method": "post", "contentType": "application/json", "payload": JSON.stringify(payload), "muteHttpExceptions": true };
    var res = UrlFetchApp.fetch(url, options);
    
    if (res.getResponseCode() !== 200) { return "ERROR_API: " + res.getContentText(); }
    
    var json = JSON.parse(res.getContentText());
    if (json.candidates && json.candidates.length > 0) return json.candidates[0].content.parts[0].text;
    return "ERROR_API: No valid data from Gemini";
  } catch (e) { return "ERROR_API: " + e.toString(); }
}

function calculatePlanPercentages(data) {
  // ✅ อัปเดต: ดึงค่าดิบก่อน เพื่อนำมาแปลงเกณฑ์
  const getRawQ = (qNum) => {
    if (data.answersArray && data.answersArray.length >= qNum) {
      return Number(data.answersArray[qNum - 1]) || 0;
    }
    return Number(data['q' + qNum]) || Number(data['Q' + qNum]) || Number(data[qNum]) || 0;
  };

  const calcCFA = (weightsObj) => {
    let earnedScore = 0;
    let maxPossibleScore = 0;
    
    for (const [qStr, weight] of Object.entries(weightsObj)) {
      const qNum = parseInt(qStr.replace(/\D/g, ''));
      const rawVal = getRawQ(qNum);
      // ✅ กรองให้เหลือเฉพาะข้อที่ตอบ 4 หรือ 5
      const qVal = (rawVal === 4 || rawVal === 5) ? rawVal : 0;
      
      const absWeight = Math.abs(weight);
      
      if (qVal > 0) { 
        if (weight < 0) {
          earnedScore += ((6 - qVal) * absWeight); // กลับด้านคะแนนสำหรับ Negative Weight
        } else {
          earnedScore += (qVal * absWeight);
        }
      } else {
        // หากตอบ 1, 2, 3 จะถือว่าคะแนนที่ได้สำหรับน้ำหนักนี้คือ 0
        earnedScore += 0; 
      }
      maxPossibleScore += (5 * absWeight); // คะแนนเต็มรายข้อคือ 5
    }
    return maxPossibleScore > 0 ? (earnedScore / maxPossibleScore) * 100 : 0;

// 🌟 ฟังก์ชันที่ 2: คำนวณเปอร์เซ็นต์ความถนัด 8 ด้าน สำหรับแสดงผลในกราฟ 
function calculateAptitudePercentages(scores) {
  const aptitudes = [
    { name: "ด้านภาษา", score: scores["ด้านภาษา"], max: 40 }, 
    { name: "ด้านตรรกะและคณิตศาสตร์", score: scores["ด้านตรรกะและคณิตศาสตร์"], max: 40 }, 
    { name: "ด้านมิติสัมพันธ์", score: scores["ด้านมิติสัมพันธ์"], max: 40 }, 
    { name: "ด้านร่างกายและการเคลื่อนไหว", score: scores["ด้านร่างกายและการเคลื่อนไหว"], max: 40 }, 
    { name: "ด้านดนตรี", score: scores["ด้านดนตรี"], max: 40 }, 
    { name: "ด้านมนุษยสัมพันธ์", score: scores["ด้านมนุษยสัมพันธ์"], max: 40 }, 
    { name: "ด้านเข้าใจตนเอง", score: scores["ด้านเข้าใจตนเอง"], max: 40 }, 
    { name: "ด้านธรรมชาติวิทยา", score: scores["ด้านธรรมชาติวิทยา"], max: 40 } 
  ];

  let results = aptitudes.map(apt => {
    return {
      name: apt.name,
      percentage: Math.round(Math.min(100, Math.max(0, (apt.score / apt.max) * 100)))
    };
  });

  results.sort((a, b) => b.percentage - a.percentage);
  return results;
}
