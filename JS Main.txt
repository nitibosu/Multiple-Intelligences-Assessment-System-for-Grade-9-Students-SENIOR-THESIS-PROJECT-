  class AssessmentState {
    constructor(totalQuestions) { this.totalQuestions = totalQuestions; this.answers = {}; this.listeners = []; }
    subscribe(listener) { this.listeners.push(listener); }
    setAnswer(questionId, score) {
      this.answers[questionId] = parseInt(score);
      const box = document.getElementById(`box_${questionId}`);
      if(box) box.classList.remove('unanswered-error');
      this.notifyUI(); 
    }
    get progress() { return (Object.keys(this.answers).length / this.totalQuestions) * 100; }
    get answeredCount() { return Object.keys(this.answers).length; }
    getMissingQuestions(allQuestionsArray) { return allQuestionsArray.filter(q => !this.answers[q.id]).map(q => q.id); }
    notifyUI() { this.listeners.forEach(listener => listener(this)); }
    reset() { this.answers = {}; this.notifyUI(); }
  }

  function isNameValid(name) {
    name = name.trim();
    if (name.length < 2) return { valid: false, message: "ชื่อสั้นเกินไป" };
    if (!/^[A-Za-zก-ฮะ-์ฤฤๅฦฦๅ\s]+$/.test(name)) return { valid: false, message: "ไม่อนุญาตให้ใช้ตัวเลขหรือสัญลักษณ์พิเศษ" };
    if (/[a-z][A-Z][a-z][A-Z]/.test(name)) return { valid: false, message: "รูปแบบตัวพิมพ์เล็ก-ใหญ่ผิดปกติ" };

    const noSpaceName = name.replace(/\s/g, '').toLowerCase();
    const uniqueChars = new Set(noSpaceName.split('')).size;
    if (uniqueChars === 1) return { valid: false, message: "ไม่อนุญาตให้พิมพ์ตัวอักษรเดียวกันซ้ำๆ" };
    if (/(.)\1{2,}/.test(name)) return { valid: false, message: "ตรวจพบการพิมพ์ตัวอักษรซ้ำกันผิดปกติ" };

    const kbRows = [
      "qwertyuiop", "asdfghjkl", "zxcvbnm", 
      "ไำพะีรนยบลฃ", "ฟหกดเ้่าสวง", "ผปแอิืทมใฝ", 
      "abcdefghijklmnopqrstuvwxyz", 
      "กขคฆงจฉชซฌญฎฏฐฑฒณดตถทธนบปผฝพฟภมยรลวศษสหฬอฮ" 
    ];

    const words = name.toLowerCase().split(/\s+/);
    for (let word of words) {
        if (word.length >= 3) {
            let isMashing = kbRows.some(row => row.includes(word) || row.includes(word.split('').reverse().join('')));
            if (isMashing) {
                return { valid: false, message: `ตรวจพบการรูดแป้นพิมพ์ (ไม่อนุญาตให้ใช้คำว่า "${word}")` };
            }
        }
        if (/^[a-z]+$/.test(word)) {
            if (!/[aeiouy]/.test(word) && word.length > 2) return { valid: false, message: "ชื่อภาษาอังกฤษต้องมีสระผสมอยู่ด้วย" };
            if (/[^aeiouy]{4,}/.test(word)) return { valid: false, message: "พิมพ์มั่ว (พยัญชนะอังกฤษติดกัน 4 ตัว)" };
            if (/[aeiouy]{4,}/.test(word)) return { valid: false, message: "พิมพ์มั่ว (สระอังกฤษติดกัน 4 ตัว)" };
        }
        else if (/^[ก-ฮะ-์ฤฤๅฦฦๅ]+$/.test(word)) {
            if (/^[ะาำิีึืุู่้๊๋์ๆฯ]/.test(word)) return { valid: false, message: "คำไทยห้ามขึ้นต้นด้วยสระหรือวรรณยุกต์" };
            if (/[เแโใไ]{2,}/.test(word)) return { valid: false, message: "สระหน้าซ้อนกันผิดปกติ" };
            if (/[ะาำิีึืุู่้๊๋์]{3,}/.test(word)) return { valid: false, message: "สระหรือวรรณยุกต์ซ้อนกันมากเกินไป" };
            let hasThaiVowel = /[ะาำิีึืุู่้๊๋์เแโใไฤฤๅอวรย]/.test(word);
            if (word.length >= 4 && !hasThaiVowel) {
                return { valid: false, message: "พิมพ์มั่ว (ไม่มีสระหรือตัวควบกล้ำที่ถูกต้อง)" };
            }
        }
    }
    if (noSpaceName.length >= 4) {
        let isMashingAll = kbRows.some(row => row.includes(noSpaceName) || row.includes(noSpaceName.split('').reverse().join('')));
        if (isMashingAll) return { valid: false, message: "ตรวจพบการรูดแป้นพิมพ์" };
    }
    return { valid: true, message: "pass" };
  }

  function isStrictEmailValid(email) {
    if (!email) return false;
    email = email.toLowerCase().trim();
    if (!/^[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}$/.test(email)) return false;
    const [localPart, domain] = email.split('@');
    if (localPart.length < 3) return false;
    if (/(.)\1{2,}/.test(localPart)) return false; 
    const kbRows = ["qwertyuiop", "asdfghjkl", "zxcvbnm", "abcdefghijklmnopqrstuvwxyz", "0123456789", "9876543210"];
    let isEmailMashing = kbRows.some(row => row.includes(localPart) || row.includes(localPart.split('').reverse().join('')));
    if (isEmailMashing) return false;

    if (domain !== 'gmail.com') {
        if (domain.startsWith('gma') && domain.endsWith('.com')) return false;
        if (domain.startsWith('g') && domain.endsWith('l.com') && domain.length >= 8 && domain.length <= 10) return false;
        if (['gmail.con', 'gmail.co', 'gmai.com', 'gmail.in.th'].includes(domain)) return false;
    }
    if (domain !== 'hotmail.com' && domain !== 'hotmail.co.th') {
        if (domain.startsWith('hotm') && domain.endsWith('.com')) return false;
        if (domain.startsWith('hot') && domain.endsWith('l.com') && domain.length >= 10 && domain.length <= 12) return false;
        if (['hotmail.con', 'hotmail.co', 'hotmai.com'].includes(domain)) return false;
    }
    if (['yaho.com', 'yahooo.com', 'outlok.com', 'outlook.con'].includes(domain)) return false;
    const fakeDomains = ['mailinator.com', '10minutemail.com', 'tempmail.com', 'guerrillamail.com', 'yopmail.com', 'throwawaymail.com', 'temp-mail.org', 'nada.ltd', 'mohmal.com', 'test.com', 'example.com', 'asd.com'];
    if (fakeDomains.includes(domain)) return false;
    const validEndings = ['.com', '.net', '.org', '.th', '.edu', '.info', '.ac.th', '.go.th'];
    return validEndings.some(ending => domain.endsWith(ending));
  }

  function handleStudentLogin() {
    try {
      let isValid = true;
      const nameInput = document.getElementById('userName');
      const emailInput = document.getElementById('email');
      if(!nameInput || !emailInput) return Swal.fire('ข้อผิดพลาด', 'ไม่พบช่องกรอกข้อมูล', 'error');

      const name = nameInput.value.trim();
      const email = emailInput.value.trim();
      
      const grpNameEl = document.getElementById('grpName');
      const grpEmailEl = document.getElementById('grpEmail');
      const nameErrEl = document.getElementById('nameErrorMsg');
      const emailErrEl = document.getElementById('emailErrorMsg');
      
      if(grpNameEl) grpNameEl.classList.remove('error');
      if(grpEmailEl) grpEmailEl.classList.remove('error');

      if (!name) {
        if(nameErrEl) nameErrEl.innerText = "กรุณากรอกชื่อผู้ใช้";
        if(grpNameEl) grpNameEl.classList.add('error'); 
        isValid = false; 
      } else {
        const nameCheck = isNameValid(name);
        if (!nameCheck.valid) {
          if(nameErrEl) nameErrEl.innerText = nameCheck.message;
          if(grpNameEl) grpNameEl.classList.add('error');
          isValid = false; 
        }
      }
      
      if (!email) {
        if(emailErrEl) emailErrEl.innerText = "กรุณากรอกอีเมล";
        if(grpEmailEl) grpEmailEl.classList.add('error'); 
        isValid = false; 
      } else if (!isStrictEmailValid(email)) { 
        if(emailErrEl) emailErrEl.innerText = "อีเมลไม่ถูกต้อง (ไม่อนุญาตให้อีเมลขยะ, ตัวเลขเรียงกัน หรือพิมพ์มั่ว)";
        if(grpEmailEl) grpEmailEl.classList.add('error'); 
        isValid = false; 
      }

      if (!isValid) return;

      userData.userName = name; 
      userData.email = email;
      
      const welcomeEl = document.getElementById('welcomeName');
      const displayEl = document.getElementById('displayName');
      const navBtn = document.getElementById('navLogoutBtn');

      if(welcomeEl) welcomeEl.innerText = name;
      if(displayEl) displayEl.innerText = name;
      if(navBtn) navBtn.style.display = 'block';

      switchScreen('loginSection', 'fullPageLoader');
      setSystemLoader('กำลังเตรียมแบบประเมิน...', 'กรุณารอสักครู่ ระบบกำลังจัดเตรียมชุดข้อคำถามพหุปัญญา');
  
      google.script.run
        .withSuccessHandler(renderQuestions)
        .withFailureHandler(err => {
          switchScreen('fullPageLoader', 'loginSection');
          Swal.fire('Error', err.message, 'error');
        })
        .getQuestionsFromDatabase();
    } catch(e) {
      Swal.fire('Error', e.message, 'error');
    }
  }

      });
      const firstBox = document.getElementById(`box_${missingIds[0]}`);
      if(firstBox) window.scrollTo({ top: firstBox.offsetTop - 120, behavior: 'smooth' });
      
      return Swal.fire({
        html: `
          <div style="text-align: center; padding: 10px 5px;">

            </div>

              ดูเหมือนว่าคุณจะข้ามไปบางข้อนะ<br>

            </p>
          </div>
        `,
        confirmButtonText: 'ตกลง (ไปตอบเพิ่ม)',
        confirmButtonColor: '#f59e0b',
        customClass: { popup: 'glass-popup', confirmButton: 'swal-confirm-custom' }
      });
    }

    switchScreen('assessmentSection', 'fullPageLoader');
    setSystemLoader('กำลังวิเคราะห์ข้อมูล', 'กรุณารอสักครู่');

    let sortedKeys = Object.keys(appState.answers).sort((a, b) => {
        let numA = parseInt(a.toString().replace(/\D/g, '')); 
        let numB = parseInt(b.toString().replace(/\D/g, ''));
        return numA - numB;
    });
    let answersArray = [];
    sortedKeys.forEach(key => answersArray.push(appState.answers[key]));

    google.script.run
      .withSuccessHandler(showResult)
      .withFailureHandler(err => { 
        switchScreen('fullPageLoader', 'assessmentSection'); 
        Swal.fire('Error', err.message, 'error'); 
      })
      .processAssessment({ 
          userName: userData.userName, 
          email: userData.email, 
          answersArray: answersArray, 
          ...appState.answers 
      });
  }

    const setText = (id, text) => { const el = document.getElementById(id); if(el) el.innerText = text; };
    
    setText('resName', userData.userName);
    const d = new Date(); setText('resDate', `${d.getDate()}/${d.getMonth()+1}/${d.getFullYear()} เวลา ${d.getHours()}:${String(d.getMinutes()).padStart(2,'0')} น.`);
    setText('resDomName', sortedScores[0].name);
    setText('resDomPercent', sortedScores[0].percentage + '%');

    // --- อัปเดตข้อมูล Podium Top 3 ---
    if (response.prediction && response.prediction.length >= 3) {
      const topPlans = response.prediction;
      
      document.getElementById('planName1').innerText = topPlans[0].plan;
      document.getElementById('planPct1').innerText = topPlans[0].percentage + '%';

      document.getElementById('planPct3').innerText = topPlans[2].percentage + '%';

    }

    let barsHtml = '';
    sortedScores.forEach((item, index) => {

      barsHtml += `
        <div class="score-row">
          <div class="score-label"><span>${index+1}. ${item.name}</span><span>${item.percentage}%</span></div>
          <div class="score-bar-bg"><div class="score-bar-fill ${colorClass}" style="width: ${item.percentage}%"></div></div>

    });
    const container = document.getElementById('scoreBarsContainer');

    const ai = response.aiData || {};
    const formatAIText = (text) => {
        if(!text || text === "-") return '-';
        let cleanText = text.replace(/[*#_.]/g, ''); 
        return cleanText.replace(/\n+/g, '<br>').trim();
    };

    const summaryEl = document.getElementById('aiSummary');
    if(summaryEl) {
      if (ai.summary && ai.summary.includes("⚠️")) {
          summaryEl.innerHTML = `<span class="text-danger fw-bold">${ai.summary}</span>`;
      } else {
          summaryEl.innerHTML = formatAIText(ai.summary);
      }
    }
    
    const setHtml = (id, text) => { const el = document.getElementById(id);
    if(el) el.innerHTML = formatAIText(text); };
    
    setHtml('aiAcademic', ai.academic);
    setHtml('aiVocational', ai.vocational);
    setHtml('aiLearning', ai.learningStyle);
    setHtml('aiDevText', ai.development);

    const careerContainer = document.getElementById('aiCareer');
    if (careerContainer && ai.career && ai.career !== "-") {
       let cleanCareer = ai.career.replace(/[*#_.]/g, '');
       let careers = cleanCareer.split(',').map(c => c.trim()).filter(c => c);
       careerContainer.innerHTML = careers.map(c => `<span class="tag-item">${c}</span>`).join('');
    } else if (careerContainer) {
       careerContainer.innerHTML = "-";
    }
    
    window.scrollTo(0, 0);
  }

function downloadImage() {
    const element = document.getElementById('pdfContentArea');
    const actionButtons = element.querySelector('[data-html2canvas-ignore]');
    
    Swal.fire({
      title: 'กำลังดาวน์โหลดผลลัพธ์...',
      text: 'กรุณารอสักครู่...',
      allowOutsideClick: false,
      didOpen: () => { Swal.showLoading(); }
    });
    
    if (actionButtons) actionButtons.style.display = 'none';
    
    // ใส่คลาสล็อกหน้าจอให้เป็นสัดส่วน Desktop
    element.classList.add('image-export-mode');
    window.scrollTo(0, 0);

    // หน่วงเวลา 0.8 วินาทีให้ CSS จัดเลย์เอาต์ให้เสร็จก่อนถ่ายภาพ
    setTimeout(() => {
      try {
        if (typeof html2canvas === 'undefined') {
          throw new Error("ไม่พบไลบรารี html2canvas");
        }
        
        html2canvas(element, {
          scale: 2, 
          useCORS: true,
          backgroundColor: '#f8fafc' 
        }).then((canvas) => {
          // คืนค่าหน้าจอกลับสู่ปกติ
          element.classList.remove('image-export-mode');
          if (actionButtons) actionButtons.style.display = 'flex';
          
          // สร้างไฟล์รูป JPG และสั่งดาวน์โหลด
          const imgData = canvas.toDataURL('image/jpeg', 0.9);
          const link = document.createElement('a');
          link.download = `MI_Report_${userData.userName || 'Student'}.jpg`;
          link.href = imgData;
          
          // ทริคเพื่อให้ทำงานได้ 100% บนทุกเบราว์เซอร์
          document.body.appendChild(link);
          link.click();
          document.body.removeChild(link);
          
          // เด้งเตือนว่าโหลดเสร็จแล้ว
          Swal.fire({
             icon: 'success',
             title: 'บันทึกสำเร็จ!',
             text: 'ดาวน์โหลดรูปภาพผลลัพธ์ลงในเครื่องของคุณแล้ว',
             timer: 2500,
             showConfirmButton: false
          });

        }).catch(err => {
          element.classList.remove('image-export-mode');
          if (actionButtons) actionButtons.style.display = 'flex';
          Swal.fire('ข้อผิดพลาด', 'ไม่สามารถบันทึกรูปภาพได้: ' + err.message, 'error');
        });

      } catch (e) {
        element.classList.remove('image-export-mode');
        if (actionButtons) actionButtons.style.display = 'flex';
        Swal.fire('ข้อผิดพลาดระบบ', e.message, 'error');
      }
    }, 800); 
  }

  function switchScreen(hideId, showId) {
    const hideEl = document.getElementById(hideId);
    const showEl = document.getElementById(showId);
    if(hideEl) hideEl.style.display = 'none';
    if(showEl) showEl.style.display = (showId === 'fullPageLoader') ? 'flex' : 'block';
  }
